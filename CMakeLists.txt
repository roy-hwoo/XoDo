cmake_minimum_required(VERSION 3.20)
project(XoDo VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "XoDo requires Windows")
endif()
add_executable(XoDo WIN32 Main.cpp Settings.cpp PinManager.cpp XoDo.rc)
target_compile_features(XoDo PRIVATE cxx_std_17)
target_compile_definitions(XoDo PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(XoDo PRIVATE user32 gdi32 shell32 comctl32 dwmapi advapi32 ole32)
if(MSVC)
    target_compile_options(XoDo PRIVATE /W4 /utf-8 /permissive-)
    set_property(TARGET XoDo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_options(XoDo PRIVATE /MANIFEST:NO)
endif()
